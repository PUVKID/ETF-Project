+(function exposeNumbers(root) {
+  function decimalParts(value, shift = 0) {
+    const match = String(value).match(/^(-?)(\d+)(?:\.(\d+))?(?:e([+-]?\d+))?$/i);
+    if (!match) return null;
+    if (Math.abs(Number(match[4] || 0)) > 400) return null;
+    const digits = match[2] + (match[3] || '');
+    const point = match[2].length + Number(match[4] || 0) + shift;
+    const integer = point <= 0 ? '0' : digits.slice(0, point).padEnd(point, '0');
+    const fraction = point < 0 ? '0'.repeat(-point) + digits : digits.slice(Math.max(0, point));
+    return {negative:match[1] === '-', integer:integer.replace(/^0+(?=\d)/, ''), fraction};
+  }
+
+  function truncateDecimal(value, places = 2, shift = 0) {
+    if (!Number.isInteger(places) || places < 0 || places > 20 || !Number.isInteger(shift) || Math.abs(shift) > 400) return '—';
+    if (typeof value === 'number' && !Number.isFinite(value)) return '—';
+    if (typeof value !== 'number' && typeof value !== 'string') return '—';
+    const parts = decimalParts(value, shift);
+    if (!parts) return '—';
+    const fraction = parts.fraction.slice(0, places).padEnd(places, '0');
+    const sign = parts.negative && /[1-9]/.test(parts.integer + fraction) ? '-' : '';
+    const integer = parts.integer.replace(/\B(?=(\d{3})+(?!\d))/g, ',');
+    return sign + integer + (places ? '.' + fraction : '');
+  }
+
+  function number(value, places = 2) {
+    return typeof value === 'number' && Number.isFinite(value) ? truncateDecimal(value, places) : '—';
+  }
+
+  function percent(value) {
+    if (typeof value !== 'number' || !Number.isFinite(value)) return '—';
+    const parts = decimalParts(value, 2);
+    const first = parts.fraction.slice(0, 1).padEnd(1, '0');
+    let scaled = BigInt(parts.integer + first);
+    if ((parts.fraction[1] || '0') >= '5') scaled += 1n;
+    const digits = String(scaled).padStart(2, '0');
+    const sign = parts.negative && scaled !== 0n ? '-' : '';
+    const integer = digits.slice(0, -1).replace(/\B(?=(\d{3})+(?!\d))/g, ',');
+    const result = sign + integer + '.' + digits.slice(-1) + '%';
+    return result === '100.0%' && value < 1 ? result + ' (반올림)' : result;
+  }
+

+  root.DisplayNumbers = api;
+  if (typeof module !== 'undefined' && module.exports) module.exports = api;

-## 보유와 거래
-
-화면은 현재 보유 ETF·수량·평균 매입가·원화 원가와 평가·미실현손익을 보여 줍니다. 대기 주문과 체결 내역을 구분하고 체결 내역은 매수·매도 필터와 페이지 이동으로 전체 기록을 읽습니다. 환전은 별도 내역입니다. 새 관측을 확보하기 전에는 거래가 0건일 수 있습니다.
-
-후보는 QQQM·SCHG·SPYM·VTI·VEA·VWO·BND·IEF·TLT·SGOV·GLD·XLK·XLF·XLV·XLE입니다. 모델 예측과 비용·현금 제약으로 자동 선택합니다. 모든 ETF 또는 토스 지원 전 종목을 검증한 목록은 아닙니다. 차입·공매도·레버리지 ETF는 사용하지 않습니다.
-
-학습용 완결 일봉과 체결용 정규장 완결 1분봉을 구분합니다. 주문 뒤 시작한 봉, 현재 정규장, 최대 20분 지연 조건을 충족해야 모의 체결합니다. 과거·장외·미래·오래된 관측은 체결로 만들지 않습니다. 무료 데이터 부재는 대기로 표시합니다. 화면의 1초 상태 갱신은 실시간 초 단위 시세를 뜻하지 않습니다.

-## 이어 학습과 성과 해석
-
-`amazon/chronos-2`의 같은 마지막 가중치를 계속 갱신합니다. 회차마다 모델을 경쟁시키거나 채택하지 않습니다. 최초 세 ETF의 고정 학습·검증·시험 구간과 기준 평가를 보존하고 신규 ETF의 과거 학습은 원래 학습 종료일 이하로 제한합니다. 최초 시험 종료 뒤 새 자료는 추가 학습이며 독립 시험 성과가 아닙니다.
-
-약 두 달 학습은 운용 계획입니다. PC와 로컬 프로세스 실행 중 반복하며 전원·절전·인터넷·프로세스 중단의 영향을 받습니다. 같은 표본의 반복 횟수와 독립 원본 수를 구분합니다.
-
-초기 20단계 시험 MAPE 원값은 추가 학습 약 0.7643%, 무변화 예측 약 0.7550%였으며 요약은 각각 0.8%, 0.8%입니다. 투자 수익 우위가 입증된 결과는 아닙니다.
-
-과거 모의 비교, 운용 후 쌓이는 전진 기록, 가정 수익률 시나리오는 다릅니다. 사전학습 데이터의 개별 자산·시점 경계는 확인되지 않았습니다. 10년은 계획 기간이며 짧은 ETF 이력을 10년 실측 성과로 만들지 않습니다. 수익·원금 보존·실거래 성능을 보장하지 않습니다.

+## LEVEL 1 · 용어와 실수하기 쉬운 구분
+
+ETF는 여러 자산에 묶어 투자할 수 있도록 거래소에 상장한 상품입니다. 이 프로젝트의 모의계좌는 실제 돈을 보내지 않고 계산으로만 보유·체결을 기록합니다. 일봉·분봉은 각각 하루·분 단위로 묶은 시세이며 완결봉은 해당 시간이 끝난 관측입니다. 가중치는 학습한 모델에 저장된 값이고 추가 학습은 그 값을 이어 갱신하는 일입니다. MAPE는 예측과 실제 값의 상대 오차를 평균한 지표로 투자 수익률과 다릅니다.
+
+**실제 토스 계좌·주문과 연결하지 않는 모의운용입니다.** 무료 자료가 없거나 오래되면 대기하며 소급 체결을 만들지 않습니다. 같은 표본 반복·추가 학습·독립 시험을 구분하고 수익·원금 보존·실거래 성능을 보장하지 않습니다.

-### LEVEL 2 · 보유와 거래
-
-이 항목의 설정과 자료 흐름은 다음 조건으로 연결됩니다. 같은 이름이 있어도 서로 다른 차수의 구성과 합치지 않습니다.
-
-화면은 현재 보유 ETF·수량·평균 매입가·원화 원가와 평가·미실현손익을 보여 줍니다. 대기 주문과 체결 내역을 구분하고 체결 내역은 매수·매도 필터와 페이지 이동으로 전체 기록을 읽습니다. 환전은 별도 내역입니다. 새 관측을 확보하기 전에는 거래가 0건일 수 있습니다.
-
-후보는 QQQM·SCHG·SPYM·VTI·VEA·VWO·BND·IEF·TLT·SGOV·GLD·XLK·XLF·XLV·XLE입니다. 모델 예측과 비용·현금 제약으로 자동 선택합니다. 모든 ETF 또는 토스 지원 전 종목을 검증한 목록은 아닙니다. 차입·공매도·레버리지 ETF는 사용하지 않습니다.
-
-학습용 완결 일봉과 체결용 정규장 완결 1분봉을 구분합니다. 주문 뒤 시작한 봉, 현재 정규장, 최대 20분 지연 조건을 충족해야 모의 체결합니다. 과거·장외·미래·오래된 관측은 체결로 만들지 않습니다. 무료 데이터 부재는 대기로 표시합니다. 화면의 1초 상태 갱신은 실시간 초 단위 시세를 뜻하지 않습니다.
-
-모델 신호→비용·현금 제약→주문→유효 시세 관측→체결→보유 갱신을 연결합니다. 거래0건도 새 관측 대기 상태일 수 있습니다.

-### LEVEL 2 · 이어 학습과 성과 해석
-
-이 항목의 설정과 자료 흐름은 다음 조건으로 연결됩니다. 같은 이름이 있어도 서로 다른 차수의 구성과 합치지 않습니다.
-
-`amazon/chronos-2`의 같은 마지막 가중치를 계속 갱신합니다. 회차마다 모델을 경쟁시키거나 채택하지 않습니다. 최초 세 ETF의 고정 학습·검증·시험 구간과 기준 평가를 보존하고 신규 ETF의 과거 학습은 원래 학습 종료일 이하로 제한합니다. 최초 시험 종료 뒤 새 자료는 추가 학습이며 독립 시험 성과가 아닙니다.
-
-약 두 달 학습은 운용 계획입니다. PC와 로컬 프로세스 실행 중 반복하며 전원·절전·인터넷·프로세스 중단의 영향을 받습니다. 같은 표본의 반복 횟수와 독립 원본 수를 구분합니다.
-
-초기 20단계 시험 MAPE 원값은 추가 학습 약 0.7643%, 무변화 예측 약 0.7550%였으며 요약은 각각 0.8%, 0.8%입니다. 투자 수익 우위가 입증된 결과는 아닙니다.
-
-과거 모의 비교, 운용 후 쌓이는 전진 기록, 가정 수익률 시나리오는 다릅니다. 사전학습 데이터의 개별 자산·시점 경계는 확인되지 않았습니다. 10년은 계획 기간이며 짧은 ETF 이력을 10년 실측 성과로 만들지 않습니다. 수익·원금 보존·실거래 성능을 보장하지 않습니다.
-
-최초 고정 학습·검증·시험과 시험 이후 추가 자료를 나누고 반복 표본과 독립 원본을 구분합니다.

-`amazon/chronos-2`의 같은 마지막 가중치를 계속 갱신합니다. 회차마다 모델을 경쟁시키거나 채택하지 않습니다. 최초 세 ETF의 고정 학습·검증·시험 구간과 기준 평가를 보존하고 신규 ETF의 과거 학습은 원래 학습 종료일 이하로 제한합니다. 최초 시험 종료 뒤 새 자료는 추가 학습이며 독립 시험 성과가 아닙니다.
-
-약 두 달 학습은 운용 계획입니다. PC와 로컬 프로세스 실행 중 반복하며 전원·절전·인터넷·프로세스 중단의 영향을 받습니다. 같은 표본의 반복 횟수와 독립 원본 수를 구분합니다.
-
-초기 20단계 시험 MAPE 원값은 추가 학습 약 0.7643%, 무변화 예측 약 0.7550%였으며 요약은 각각 0.8%, 0.8%입니다. 투자 수익 우위가 입증된 결과는 아닙니다.
-
-과거 모의 비교, 운용 후 쌓이는 전진 기록, 가정 수익률 시나리오는 다릅니다. 사전학습 데이터의 개별 자산·시점 경계는 확인되지 않았습니다. 10년은 계획 기간이며 짧은 ETF 이력을 10년 실측 성과로 만들지 않습니다. 수익·원금 보존·실거래 성능을 보장하지 않습니다.

+화면은 현재 보유 ETF·수량·평균 매입가·원화 원가와 평가·미실현손익을 보여 줍니다. 대기 주문과 체결 내역을 구분하고 체결 내역은 매수·매도 필터와 페이지 이동으로 전체 기록을 읽습니다. 환전은 별도 내역입니다. 새 관측을 확보하기 전에는 거래가 0건일 수 있습니다.
+
+후보는 QQQM·SCHG·SPYM·VTI·VEA·VWO·BND·IEF·TLT·SGOV·GLD·XLK·XLF·XLV·XLE입니다. 모델 예측과 비용·현금 제약으로 자동 선택합니다. 모든 ETF 또는 토스 지원 전 종목을 검증한 목록은 아닙니다. 차입·공매도·레버리지 ETF는 사용하지 않습니다.
+
+학습용 완결 일봉과 체결용 정규장 완결 1분봉을 구분합니다. 주문 뒤 시작한 봉, 현재 정규장, 최대 20분 지연 조건을 충족해야 모의 체결합니다. 과거·장외·미래·오래된 관측은 체결로 만들지 않습니다. 무료 데이터 부재는 대기로 표시합니다. 화면의 1초 상태 갱신은 실시간 초 단위 시세를 뜻하지 않습니다.
+
+모델 신호→비용·현금 제약→주문→유효 시세 관측→체결→보유 갱신을 연결합니다. 거래0건도 새 관측 대기 상태일 수 있습니다.

+### LEVEL 2 · 이어 학습과 성과 해석
+
+이 항목의 설정과 자료 흐름은 다음 조건으로 연결됩니다. 같은 이름이 있어도 서로 다른 차수의 구성과 합치지 않습니다.
+
+`amazon/chronos-2`의 같은 마지막 가중치를 계속 갱신합니다. 회차마다 모델을 경쟁시키거나 채택하지 않습니다. 3차에서는 최초 세 ETF의 고정 학습·검증·시험 구간과 기준 평가를 보존하고 신규 ETF의 과거 학습을 원래 학습 종료일 이하로 제한했습니다. 4차의 학습 범위는 위 표를 따릅니다. 최초 시험 종료 뒤 새 자료는 추가 학습이며 독립 시험 성과가 아닙니다.
+
+약 두 달 학습은 운용 계획입니다. PC와 로컬 프로세스 실행 중 반복하며 전원·절전·인터넷·프로세스 중단의 영향을 받습니다. 같은 표본의 반복 횟수와 독립 원본 수를 구분합니다.
+
+초기 20단계 시험 MAPE 원값은 추가 학습 약 0.7643%, 무변화 예측 약 0.7550%였으며 요약은 각각 0.8%, 0.8%입니다. 투자 수익 우위가 입증된 결과는 아닙니다.
+
+과거 모의 비교, 운용 후 쌓이는 전진 기록, 가정 수익률 시나리오는 다릅니다. 사전학습 데이터의 개별 자산·시점 경계는 확인되지 않았습니다. 10년은 계획 기간이며 짧은 ETF 이력을 10년 실측 성과로 만들지 않습니다. 수익·원금 보존·실거래 성능을 보장하지 않습니다.
+
+최초 고정 학습·검증·시험과 시험 이후 추가 자료를 나누고 반복 표본과 독립 원본을 구분합니다.
