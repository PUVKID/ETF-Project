+def parse_quote(symbol, chart, fetched_at, source_url):

+def collect_quotes(symbols, fetched_at=None, fetcher=None):
+    fetcher = fetcher or fetch_intraday
+    quotes, errors, details = {}, {}, {}
+    with ThreadPoolExecutor(max_workers=3) as executor:
+        futures = {executor.submit(fetcher, symbol): symbol for symbol in symbols}

+            except Exception as error:

+    errors = 0

+def evaluation_windows(series, split, segment, context_length, limit=48):
+    start = split[f"{segment}_start"]
+    end = split[f"{segment}_end"]
+    inputs, actual, previous, keys = [], [], [], []
+    for symbol, rows in sorted(series.items()):
+        available = [index for index, row in enumerate(rows) if start <= row["date"] <= end and index >= 32]
+        # 일정 간격 표본이며 미래 가격이나 성과를 기준으로 표본을 고르지 않습니다.
+        if len(available) > limit:
+            available = sorted({available[int(index * (len(available) - 1) / (limit - 1))] for index in range(limit)})

+            actual.append(rows[index]["close"])

+    if not inputs:

+def predict_values(pipeline, inputs, context_length, batch_size=4):
+    import numpy as np
+    predictions = pipeline.predict([np.asarray(values, dtype=np.float32) for values in inputs],
+                                   prediction_length=1, context_length=context_length, batch_size=batch_size)
+    median = min(range(len(pipeline.quantiles)), key=lambda index: abs(pipeline.quantiles[index] - 0.5))
+    return [float(prediction[0, median, 0].detach().cpu()) for prediction in predictions]

+def save_snapshot(root, category, value):
+    """내용이 같은 자료는 한 번만 저장하고 기존 파일의 무결성을 확인합니다."""
+    root = Path(root)
+    encoded = _encoded(value)
+    digest = hashlib.sha256(encoded).hexdigest()
+    relative = Path("자료스냅샷_4차") / category / (digest + ".json")
+    path = root / relative
+    path.parent.mkdir(parents=True, exist_ok=True)
+    if path.exists():
+        if hashlib.sha256(path.read_bytes()).hexdigest() != digest:
+            raise RuntimeError("보존 자료 스냅샷 해시가 일치하지 않습니다")
+    else:
+        # 완성된 임시 파일만 이동하여 쓰기 실패가 영구 해시 파일을 오염시키지 않게 합니다.
+        descriptor, temporary = tempfile.mkstemp(prefix=".스냅샷-", suffix=".tmp", dir=path.parent)
+        try:
+            with os.fdopen(descriptor, "wb") as stream:
+                stream.write(encoded)
+                stream.flush()
+                os.fsync(stream.fileno())
+            os.replace(temporary, path)
+        finally:
+            if os.path.exists(temporary):
+                os.unlink(temporary)
+    return {"path": relative.as_posix(), "sha256": digest, "bytes": len(encoded)}

+def _rows(rows, as_of, field="close"):
+    result = {}
+    for original in rows:
+        row = dict(original)
+        day = row["date"]
+        if date.fromisoformat(day).isoformat() != day or not session_completed(day, as_of):
+            raise ValueError(f"{day}: 미래·미완결 또는 잘못된 날짜의 학습자료입니다")
+        value = float(row[field])
+        if not math.isfinite(value) or value <= 0:
+            raise ValueError(f"{day}: 유효하지 않은 학습 관측값입니다")
+        row[field] = value
+        if "open" in row and (not math.isfinite(float(row["open"])) or float(row["open"]) <= 0):
+            raise ValueError(f"{day}: 유효하지 않은 시가입니다")
+        if day in result and result[day] != row:
+            raise ValueError(f"{day}: 중복 날짜의 값이 다릅니다")
+        result[day] = row
+    return result
